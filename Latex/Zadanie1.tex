% Obrázok sa vloží, len ak existuje - inak sa vysádza rámček s upozornením
% (dokument sa teda preloží aj vtedy, keď niektorý model ešte nebol spustený).
\newcommand{\safeincludegraphics}[2][]{%
    \IfFileExists{#2}{%
        \includegraphics[#1]{#2}%
    }{%
        \fbox{\parbox{0.7\textwidth}{\centering Chýbajúci obrázok: \texttt{\detokenize{#2}}}}%
    }%
}

% Skratka pre obrázok na šírku textu s popisom a menovkou
\newcommand{\obrazok}[4][\textwidth]{%
    \begin{figure}[H]
        \centering
        \safeincludegraphics[width=#1]{#2}
        \caption{#3}
        \label{#4}
    \end{figure}%
}

% Obrázok otočený na šírku, na samostatnej strane. Používa sa pre porovnávacie
% grafy priebehu trénovania (2x2 panely) - na výšku strany by boli popisy osí
% a legenda prakticky nečitateľné, otočením sa obrázok zväčší zhruba o polovicu.
\newcommand{\obrazoknasirku}[4][0.95\textheight]{%
    \begin{sidewaysfigure}
        \centering
        \safeincludegraphics[width=#1]{#2}
        \caption{#3}
        \label{#4}
    \end{sidewaysfigure}%
}
